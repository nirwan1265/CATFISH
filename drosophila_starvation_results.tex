\section*{Top Candidate Genes - Drosophila Starvation Resistance}

\subsection*{Scoring Formula}

\begin{equation}
\text{score} = \underbrace{(\text{hit}_{\text{gwas}} + \text{hit}_{\text{magma}} + \text{hit}_{\text{pathway}})}_{\text{layer indicators (0 or 1)}} + 0.2 \cdot (-\log_{10} p_{\text{magma}}) + 0.1 \cdot (-\log_{10} p_{\text{gwas}}) + 0.1 \cdot (-\log_{10} p_{\text{pathway}})
\end{equation}
